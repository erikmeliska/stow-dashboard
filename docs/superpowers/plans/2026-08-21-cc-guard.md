# cc-guard Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A standalone, zero-dependency Claude Code PreToolUse hook that blocks destructive shell commands and audits every guard decision.

**Architecture:** One tiny Node ESM package installed into `~/.claude/settings.json` as a `PreToolUse` hook matching `Bash`. On each Bash call it reads the hook JSON from stdin, evaluates the command against a rule set (deny / warn / allow), denies via the documented JSON contract, and appends every non-allow decision to a local JSONL audit log. Pure rule logic is isolated from I/O so it is unit-testable without a real Claude session.

**Tech Stack:** Node.js ≥ 24, ESM, built-in `node:test` + `node:assert` (no third-party test runner), zero runtime dependencies.

## Global Constraints

- Runtime: **Node.js ≥ 24**, ESM only (`"type": "module"`). Verbatim from spec §3.
- **Zero third-party dependencies** (runtime and dev). Testing uses built-in `node:test`. Verbatim from spec §2 ("one file, zero deps").
- Repo location: **`/Users/ericsko/Projekty/_Bizz/TriSoft/cc-guard`** (standalone, per spec §2 hybrid decision).
- **PreToolUse deny contract (primary):** exit code `0` + stdout JSON
  `{"hookSpecificOutput":{"hookEventName":"PreToolUse","permissionDecision":"deny","permissionDecisionReason":"<reason>"}}`.
  Allow = exit `0`, no JSON. (Exit `2` + stderr is the legacy fallback; not used.)
- Hook stdin fields used: `tool_name`, `tool_input.command`, `cwd`, `session_id`, `transcript_path`.
- Audit path: `~/.claude/cc-guard/audit.jsonl`, overridable by env `CC_GUARD_AUDIT`.
- Override (one-shot allow of a deny match): env `CC_GUARD_OVERRIDE=1`, audited as `override`.
- Settings path for install: `~/.claude/settings.json`, overridable by env `CC_GUARD_SETTINGS` (used by tests).
- Guard only acts on `tool_name === "Bash"`; all other tools pass through with exit 0 and no audit.

```javascript
import { readFile } from 'node:fs/promises';

// Matches an rm target that is dangerous: /, ~, $HOME, a bare * or /*,
// or root-level absolute paths. Deliberately does NOT match ./foo,
// relative names, or node_modules — those are ordinary deletes.
export const DANGEROUS_PATH_RE =
  /(?:^|\s)(?:-[a-zA-Z]+\s+)*(?:\/(?:\s|$|\*)|\/\*|~(?:\/|\s|$)|\$HOME(?:\/|\s|$)|\*(?:\s|$))/;

export async function loadRules(path) {
  const url = path ? path : new URL('../guard.rules.json', import.meta.url);
  const data = JSON.parse(await readFile(url, 'utf8'));
  const rules = data.rules.map((r) => ({
    id: r.id,
    re: new RegExp(r.source, r.flags || ''),
    severity: r.severity,
    description: r.description,
    target: r.target,
  }));
  const allowlist = (data.allowlist || []).map((s) => new RegExp(s));
  return { rules, allowlist };
}

export function evaluate(command, ruleset) {
  const cmd = String(command || '');
  for (const rx of ruleset.allowlist || []) {
    if (rx.test(cmd)) return { action: 'allow', rule: null };
  }
  // deny rules first, then warn — first match wins within each severity.
  for (const severity of ['deny', 'warn']) {
    for (const rule of ruleset.rules) {
      if (rule.severity !== severity) continue;
      if (!rule.re.test(cmd)) continue;
      if (rule.target === 'dangerous-path' && !DANGEROUS_PATH_RE.test(cmd)) continue;
      return { action: severity, rule };
    }
  }
  return { action: 'allow', rule: null };
}
```

```javascript
import { mkdir, appendFile } from 'node:fs/promises';
import { homedir } from 'node:os';
import { join, dirname } from 'node:path';

export function auditPath(env = process.env) {
  if (env.CC_GUARD_AUDIT) return env.CC_GUARD_AUDIT;
  return join(homedir(), '.claude', 'cc-guard', 'audit.jsonl');
}

export async function appendAudit(event, opts = {}) {
  const path = opts.path || auditPath(opts.env || process.env);
  const line = JSON.stringify({ ts: new Date().toISOString(), ...event }) + '\n';
  await mkdir(dirname(path), { recursive: true });
  await appendFile(path, line, 'utf8');
  return path;
}
```

```javascript
import { evaluate, loadRules } from './rules.mjs';
import { appendAudit } from './audit.mjs';

export function decide(event, ruleset, env = process.env) {
  if (event?.tool_name !== 'Bash') {
    return { permission: 'allow', reason: null, audit: null };
  }
  const command = event.tool_input?.command ?? '';
  const { action, rule } = evaluate(command, ruleset);
  const base = {
    command,
    rule: rule?.id ?? null,
    session_id: event.session_id ?? null,
    cwd: event.cwd ?? null,
  };

  if (action === 'deny') {
    if (env.CC_GUARD_OVERRIDE === '1') {
      return { permission: 'allow', reason: null, audit: { ...base, action: 'override' } };
    }
    return {
      permission: 'deny',
      reason: `cc-guard: ${rule.description} (rule ${rule.id})`,
      audit: { ...base, action: 'deny' },
    };
  }
  if (action === 'warn') {
    return { permission: 'allow', reason: null, audit: { ...base, action: 'warn' } };
  }
  return { permission: 'allow', reason: null, audit: null };
}

async function readAll(stream) {
  let data = '';
  for await (const chunk of stream) data += chunk;
  return data;
}

export async function runHook({ stdin, stdout }, env = process.env) {
  let event;
  try {
    event = JSON.parse(await readAll(stdin));
  } catch {
    return 0; // malformed input: make no decision, let normal flow proceed
  }
  const ruleset = await loadRules();
  const d = decide(event, ruleset, env);
  if (d.audit) await appendAudit(d.audit, { env });
  if (d.permission === 'deny') {
    stdout.write(JSON.stringify({
      hookSpecificOutput: {
        hookEventName: 'PreToolUse',
        permissionDecision: 'deny',
        permissionDecisionReason: d.reason,
      },
    }));
  }
  return 0;
}
```

```javascript
import { readFile, writeFile, copyFile } from 'node:fs/promises';
import { homedir } from 'node:os';
import { join, resolve, dirname } from 'node:path';
import { fileURLToPath } from 'node:url';

const DEFAULT_BIN = resolve(dirname(fileURLToPath(import.meta.url)), '..', 'bin', 'cc-guard.mjs');

export function settingsPath(env = process.env) {
  return env.CC_GUARD_SETTINGS || join(homedir(), '.claude', 'settings.json');
}

function hookCommand(binPath) {
  return `node ${binPath || DEFAULT_BIN} hook`;
}

async function readSettings(path) {
  try {
    return JSON.parse(await readFile(path, 'utf8'));
  } catch {
    return {};
  }
}

function hasEntry(list, cmd) {
  return list.some((h) => (h.hooks || []).some((x) => x.command === cmd));
}

export async function installHook(opts = {}) {
  const path = opts.path || settingsPath();
  const cmd = hookCommand(opts.binPath);
  const settings = await readSettings(path);
  settings.hooks ||= {};
  settings.hooks.PreToolUse ||= [];
  if (hasEntry(settings.hooks.PreToolUse, cmd)) return { changed: false, path };
  settings.hooks.PreToolUse.push({
    matcher: 'Bash',
    hooks: [{ type: 'command', command: cmd }],
  });
  try { await copyFile(path, path + '.bak'); } catch { /* no prior file */ }
  await writeFile(path, JSON.stringify(settings, null, 2) + '\n', 'utf8');
  return { changed: true, path };
}

export async function uninstallHook(opts = {}) {
  const path = opts.path || settingsPath();
  const cmd = hookCommand(opts.binPath);
  const settings = await readSettings(path);
  const list = settings.hooks?.PreToolUse;
  if (!Array.isArray(list)) return { changed: false, path };
  const next = list
    .map((h) => ({ ...h, hooks: (h.hooks || []).filter((x) => x.command !== cmd) }))
    .filter((h) => (h.hooks || []).length > 0);
  if (next.length === list.length) return { changed: false, path };
  settings.hooks.PreToolUse = next;
  await writeFile(path, JSON.stringify(settings, null, 2) + '\n', 'utf8');
  return { changed: true, path };
}
```

```javascript
#!/usr/bin/env node
import { runHook } from '../src/guard.mjs';
import { installHook, uninstallHook, settingsPath } from '../src/install.mjs';
import { loadRules, evaluate } from '../src/rules.mjs';
import { auditPath } from '../src/audit.mjs';

export async function cli(argv, io = {}, env = process.env) {
  const stdout = io.stdout || process.stdout;
  const stderr = io.stderr || process.stderr;
  const [sub, ...rest] = argv;

  switch (sub || 'hook') {
    case 'hook':
      return runHook({ stdin: io.stdin || process.stdin, stdout }, env);
    case 'install': {
      const r = await installHook();
      stdout.write(`${r.changed ? 'Installed' : 'Already installed'}: ${r.path}\n`);
      return 0;
    }
    case 'uninstall': {
      const r = await uninstallHook();
      stdout.write(`${r.changed ? 'Removed' : 'Nothing to remove'}: ${r.path}\n`);
      return 0;
    }
    case 'check': {
      const rs = await loadRules();
      const { action, rule } = evaluate(rest.join(' '), rs);
      stdout.write(`${action.toUpperCase()}${rule ? ` (${rule.id}: ${rule.description})` : ''}\n`);
      return 0;
    }
    case 'doctor': {
      const rs = await loadRules();
      stdout.write(`cc-guard\n  rules: ${rs.rules.length}\n  settings: ${settingsPath(env)}\n  audit: ${auditPath(env)}\n`);
      return 0;
    }
    default:
      stderr.write(`Unknown subcommand: ${sub}\n`);
      return 1;
  }
}

const invokedDirectly = process.argv[1] && import.meta.url === `file://${process.argv[1]}`;
if (invokedDirectly) {
  cli(process.argv.slice(2)).then((code) => { process.exitCode = code; });
}
```
